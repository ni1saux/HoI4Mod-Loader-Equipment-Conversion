# HOI4 Loader Equipment Conversion: the gameplay mod for the loader's equipment_conversion tools.
# The loader does nothing on its own; this mod switches its tools on through global flags that
# common/scripted_effects/hoi4_loader_equipment_conversion_effects.txt sets from these rules: at
# game start (on_actions), or from the console for an existing save
# (event hoi4_loader_equipment_conversion.1; .2 opts out) (MODDING.md "Converting earlier generations"). With them, a production line's "Convert from
# Stockpile" also takes equipment of earlier generations (infantry_equipment_0 and _1 into _2,
# older chassis, airframes and hulls into newer ones). Without the loader the rules show but do
# nothing.
# Rule-level desc is not accepted ("Unexpected token: desc"); descriptions are on the options.
hoi4_loader_conversion_generations = {
	name = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS"
	group = "RULE_GROUP_HOI4_LOADER_CONVERSION"
	default = {
		name = ALLOWED
		text = "RULE_OPTION_ALLOWED"
		desc = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS_ALLOWED_DESC"
	}
	option = {
		name = BLOCKED
		text = "RULE_OPTION_BLOCKED"
		desc = "RULE_HOI4_LOADER_CONVERSION_GENERATIONS_BLOCKED_DESC"
		allow_achievements = yes
	}
}
